const { Client, GatewayIntentBits, EmbedBuilder, ActionRowBuilder, ButtonBuilder, ButtonStyle } = require('discord.js');

const client = new Client({ intents: [GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent] });

client.on('ready', () => {
    console.log(`Logged in as ${client.user.tag}!`);
});

// Command to send the initial embed with the button
client.on('messageCreate', async (message) => {
    if (message.content === '!setupbutton') {
        const embed = new EmbedBuilder()
            .setTitle('Claim Your Custom Role')
            .setDescription('Click the button below to automatically create and receive your new role!')
            .setColor(0x0099FF);

        const row = new ActionRowBuilder().addComponents(
            new ButtonBuilder()
                .setCustomId('create_new_role_btn')
                .setLabel('Create & Get Role')
                .setStyle(ButtonStyle.Primary)
        );

        await message.channel.send({ embeds: [embed], components: [row] });
    }
});

// Listener to handle the button click interaction
client.on('interactionCreate', async (interaction) => {
    if (!interaction.isButton()) return;

    if (interaction.customId === 'create_new_role_btn') {
        // Defer reply to give the bot time to process (prevents "Interaction Failed" errors)
        await interaction.deferReply({ ephemeral: true });

        const guild = interaction.guild;
        const member = interaction.member;
        const roleName = "New Custom Role"; // Set your desired new role name here

        try {
            // Check if the role already exists in the server to prevent duplicates
            let role = guild.roles.cache.find(r => r.name === roleName);

            // If the role doesn't exist, create it
            if (!role) {
                role = await guild.roles.create({
                    name: roleName,
                    color: '#FF5733', // Set your preferred color
                    reason: 'Automated role creation via button click',
                });
            }

            // Assign the role to the member who clicked the button
            if (!member.roles.cache.has(role.id)) {
                await member.roles.add(role);
                await interaction.editReply({ content: `Successfully created and added the **${roleName}** role to you!` });
            } else {
                await interaction.editReply({ content: `You already have the **${roleName}** role.` });
            }

        } catch (error) {
            console.error(error);
            await interaction.editReply({ content: 'An error occurred while managing roles. Ensure the bot has "Manage Roles" permissions.' });
        }
    }
});

client.login('YOUR_BOT_TOKEN');
